Run this Python code via Bash (using `python` or `python3`):

```python
import os, sys
from datetime import datetime

# pip install python-docx if needed
try:
    from docx import Document
    from docx.shared import Pt, RGBColor, Inches
    from docx.enum.text import WD_ALIGN_PARAGRAPH
except ImportError:
    os.system(sys.executable + " -m pip install python-docx --quiet")
    from docx import Document
    from docx.shared import Pt, RGBColor, Inches
    from docx.enum.text import WD_ALIGN_PARAGRAPH

# ── fill these in from the generated entry ──────────────────────────────────
entry_date    = "<YYYY-MM-DD>"          # e.g. "2026-05-03"
title         = "<Session Title>"       # e.g. "Refactor CSV Parser"
session_type  = "<Session Type>"

import subprocess
try:
    developer = subprocess.check_output(["git", "config", "user.name"], text=True).strip()
except Exception:
    developer = os.environ.get("USERNAME") or os.environ.get("USER") or "Unknown"
objective     = "<Objective text>"
original_input = """$ARGUMENTS"""

# structured sections as list of (heading, body) tuples; body may be plain text or a list of row-lists for tables
sections = [
    # ("Section Heading", "body text  OR  [['col1','col2'], ['r1c1','r1c2'], ...]")
]
# ────────────────────────────────────────────────────────────────────────────

out_dir = r"C:\Users\ninag\Documents\CS_Notebook"
os.makedirs(out_dir, exist_ok=True)

filename = f"data_{entry_date}_time_{datetime.now().strftime('%H-%M-%S')}.docx"
filepath = os.path.join(out_dir, filename)

doc = Document()

# Title
p = doc.add_heading(title, level=0)
p.alignment = WD_ALIGN_PARAGRAPH.CENTER

# Metadata block
meta = doc.add_paragraph()
meta.add_run("Date: ").bold = True
meta.add_run(entry_date + "    ")
meta.add_run("Developer: ").bold = True
meta.add_run(developer + "    ")
meta.add_run("Type: ").bold = True
meta.add_run(session_type)

doc.add_heading("Objective", level=1)
doc.add_paragraph(objective)

# Add each section
for heading, body in sections:
    doc.add_heading(heading, level=1)
    if isinstance(body, list):
        tbl = doc.add_table(rows=1, cols=len(body[0]))
        tbl.style = "Table Grid"
        hdr_cells = tbl.rows[0].cells
        for i, h in enumerate(body[0]):
            hdr_cells[i].text = h
            hdr_cells[i].paragraphs[0].runs[0].bold = True
        for row in body[1:]:
            row_cells = tbl.add_row().cells
            for i, val in enumerate(row):
                row_cells[i].text = str(val)
    else:
        doc.add_paragraph(str(body))

# Original input — always last
doc.add_heading("Original Input", level=1)
p = doc.add_paragraph()
p.add_run("The following raw notes were used to generate this entry:").italic = True
doc.add_paragraph(original_input).style = "No Spacing"

doc.save(filepath)
print(f"Saved: {filepath}")
```
